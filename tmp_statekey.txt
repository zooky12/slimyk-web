using System;
using System.Collections.Generic;
using SlimeGrid.Logic;

namespace SlimeGrid.Tools.Solver
{
    public readonly struct StateKey : IEquatable<StateKey>
    {
        public readonly ulong A;
        public readonly ulong B;
        public StateKey(ulong a, ulong b) { A = a; B = b; }
        public bool Equals(StateKey other) => A == other.A && B == other.B;
        public override bool Equals(object obj) => obj is StateKey k && Equals(k);
        public override int GetHashCode() => (int)(A ^ (B * 0x9E3779B97F4A7C15UL));
    }

    // Precomputed level context data used to compute toggle parity and canonical ordering.
    public sealed class LevelContext
    {
        public readonly Grid2D Grid;
        public readonly ToggleCell[] Togglables; // row-major order

        public LevelContext(Grid2D grid, ToggleCell[] togglables)
        { Grid = grid; Togglables = togglables; }
    }

    public readonly struct ToggleCell
    {
        public readonly V2 Pos;
        public readonly bool ByButton;
        public readonly bool ByEntity;
        public readonly bool ByPlayer;
        public ToggleCell(V2 pos, bool byButton, bool byEntity, bool byPlayer)
        { Pos = pos; ByButton = byButton; ByEntity = byEntity; ByPlayer = byPlayer; }
    }

    public static class StateHasher
    {
        const ulong FNV_OFFSET = 1469598103934665603UL;
        const ulong FNV_PRIME = 1099511628211UL;

        static void Mix(ref ulong h, ulong v)
        { h ^= v; h *= FNV_PRIME; }

        public static LevelContext BuildLevelContext(Grid2D grid)
        {
            var list = new List<ToggleCell>();
            for (int y = 0; y < grid.H; y++)
            for (int x = 0; x < grid.W; x++)
            {
                var p = new V2(x, y);
                ref var c = ref grid.CellRef(p);
                var traits = TileTraits.For(c.Type).Active;
                bool byButton = (traits & Traits.ToggleableByButton) != 0 || (traits & Traits.UntoggleableByButton) != 0;
                bool byEntity = (traits & Traits.ToggleableByEntity) != 0 || (traits & Traits.UntoggleableByEntity) != 0;
                bool byPlayer = (traits & Traits.ToggleableByPlayer) != 0 || (traits & Traits.UntoggleableByPlayer) != 0;
                if (byButton || byEntity || byPlayer)
                    list.Add(new ToggleCell(p, byButton, byEntity, byPlayer));
            }
            return new LevelContext(grid, list.ToArray());
        }

        static bool ComputeAnyButtonPressed(GameState s)
        {
            foreach (var kv in s.EntitiesById)
            {
                var e = kv.Value;
                if ((e.Traits & Traits.PressesButtons) == 0) continue;
                var tile = TileTraits.For(s.Grid.CellRef(e.Pos).Type).Active;
                if ((tile & Traits.ButtonToggle) != 0)
                    return true;
            }
            return false;
        }

        public static StateKey Compute(GameState s, LevelContext ctx)
        {
            // Two parallel FNV hashes for low collision risk.
            ulong h1 = FNV_OFFSET;
            ulong h2 = 0xCBF29CE484222325UL; // different offset

            // 1) Canonicalize entities by (Type, Pos.x, Pos.y, Orientation, Traits)
            var tmp = s.EntitiesById.Count > 0 ? new List<Entity>(s.EntitiesById.Count) : new List<Entity>();
            foreach (var kv in s.EntitiesById) tmp.Add(kv.Value);
            tmp.Sort((a, b) =>
            {
                int c = ((int)a.Type).CompareTo((int)b.Type);
                if (c != 0) return c;
                c = a.Pos.x.CompareTo(b.Pos.x); if (c != 0) return c;
                c = a.Pos.y.CompareTo(b.Pos.y); if (c != 0) return c;
                c = ((int)a.Orientation).CompareTo((int)b.Orientation); if (c != 0) return c;
                return 0; // ignore Traits for canonical ordering
            });

            // Map entity id -> canonical index
            var idToIdx = new Dictionary<int, int>(tmp.Count);
            for (int i = 0; i < tmp.Count; i++) idToIdx[tmp[i].Id] = i;

            // 2) Player core fields
            Mix(ref h1, (ulong)(uint)s.PlayerPos.x); Mix(ref h2, (ulong)(uint)s.PlayerPos.y);
            int attachedIdx = s.AttachedEntityId.HasValue && idToIdx.TryGetValue(s.AttachedEntityId.Value, out var idx) ? idx : -1;
            Mix(ref h1, (ulong)(attachedIdx + 1)); // -1 -> 0
            Mix(ref h2, (ulong)(s.EntryDir.HasValue ? (byte)s.EntryDir.Value + 1 : 0));
            // Include AnyButtonPressed and LastAnyButtonPressed (edge parity)
            Mix(ref h1, s.AnyButtonPressed ? 0xCAFED00DUL : 0xD00DFACEUL);
            Mix(ref h2, s.AnyButtonPressed ? 0x0BADC0DEUL : 0xDEADC0DEUL);
            Mix(ref h1, s.LastAnyButtonPressed ? 0xABCDEF01UL : 0x10FEDCBAUL);
            Mix(ref h2, s.LastAnyButtonPressed ? 0x13579BDFUL : 0xFDB97531UL);

            // 3) Canonical entity list
            foreach (var e in tmp)
            {
                Mix(ref h1, (ulong)(byte)e.Type);
                Mix(ref h1, (ulong)(uint)e.Pos.x); Mix(ref h1, (ulong)(uint)e.Pos.y);
                Mix(ref h1, (ulong)(byte)e.Orientation);

                Mix(ref h2, (ulong)(byte)e.Type * 1315423911UL);
                Mix(ref h2, (ulong)(((uint)e.Pos.x << 16) ^ (uint)(e.Pos.y)));
                Mix(ref h2, (ulong)(byte)e.Orientation * 1099511628211UL);
            }

            // 4) Tile signature at togglable cells (captures swap state)
            foreach (var t in ctx.Togglables)
            {
                ref var cell = ref s.Grid.CellRef(t.Pos);
                var tt = cell.Type;
                byte ori = (byte)(cell.Orientation.HasValue ? ((byte)cell.Orientation.Value + 1) : 0);
                ulong pv = ((ulong)(uint)t.Pos.x << 32) ^ (ulong)(uint)t.Pos.y;
                Mix(ref h1, pv ^ (((ulong)(byte)tt * 0x9E3779B1UL) ^ (ulong)ori));
                Mix(ref h2, pv ^ (((ulong)(byte)tt * 0xC2B2AE3DUL) ^ ((ulong)ori * 1315423911UL)));
            }

            // 5) Full-grid tile signature (strict) including orientation
            for (int y = 0; y < s.Grid.H; y++)
                for (int x = 0; x < s.Grid.W; x++)
                {
                    var p = new V2(x, y);
                    ref var c = ref s.Grid.CellRef(p);
                    var tt = c.Type;
                    byte ori = (byte)(c.Orientation.HasValue ? ((byte)c.Orientation.Value + 1) : 0);
                    ulong pv = ((ulong)(uint)x << 32) ^ (ulong)(uint)y;
                    Mix(ref h1, pv ^ (((ulong)(byte)tt * 0x85EBCA6BUL) ^ (ulong)ori));
                    Mix(ref h2, pv ^ (((ulong)(byte)tt * 0x27D4EB2FUL) ^ ((ulong)ori * 2654435761UL)));
                }

            return new StateKey(h1, h2);
        }

        // Zobrist-style hasher (order-insensitive, low allocation). Designed for speed.
        public static StateKey ComputeZobrist(GameState s, LevelContext ctx)
        {
            ulong h1 = 0x9E3779B97F4A7C15UL; // different offsets for each stream
            ulong h2 = 0xC2B2AE3D27D4EB4FUL;

            static ulong Mix64(ulong x)
            {
                x ^= x >> 33;
                x *= 0xff51afd7ed558ccdUL;
                x ^= x >> 33;
                x *= 0xc4ceb9fe1a85ec53UL;
                x ^= x >> 33;
                return x;
            }
            static void XorMix(ref ulong h, ulong v)
            {
                h ^= Mix64(v + 0x9E3779B97F4A7C15UL);
            }

            // Build canonical entity order to encode WHICH entity is attached
            var tmp = s.EntitiesById.Count > 0 ? new List<Entity>(s.EntitiesById.Count) : new List<Entity>();
            foreach (var kv in s.EntitiesById) tmp.Add(kv.Value);
            tmp.Sort((a, b) =>
            {
                int c = ((int)a.Type).CompareTo((int)b.Type);
                if (c != 0) return c;
                c = a.Pos.x.CompareTo(b.Pos.x); if (c != 0) return c;
                c = a.Pos.y.CompareTo(b.Pos.y); if (c != 0) return c;
                c = ((int)a.Orientation).CompareTo((int)b.Orientation); if (c != 0) return c;
                return 0;
            });
            var idToIdx = new Dictionary<int, int>(tmp.Count);
            for (int i = 0; i < tmp.Count; i++) idToIdx[tmp[i].Id] = i;

            // Player position + WHICH entity (canonical index) is attached + entryDir
            unchecked
            {
                ulong p = ((ulong)(uint)s.PlayerPos.x << 32) ^ (ulong)(uint)s.PlayerPos.y;
                XorMix(ref h1, p ^ 0xA5A5A5A5A5A5A5A5UL);
                XorMix(ref h2, p ^ 0x5A5A5A5A5A5A5A5AUL);
                int ed = s.EntryDir.HasValue ? ((int)s.EntryDir.Value + 1) : 0;
                int attIdx = (s.AttachedEntityId.HasValue && idToIdx.TryGetValue(s.AttachedEntityId.Value, out var idx)) ? (idx + 1) : 0;
                XorMix(ref h1, (ulong)((ed & 0xFF) | ((attIdx & 0xFFFF) << 8)));
                XorMix(ref h2, (ulong)((attIdx & 0xFFFF) | ((ed & 0xFF) << 16)));
                // Include AnyButtonPressed and LastAnyButtonPressed (edge parity)
                XorMix(ref h1, s.AnyButtonPressed ? 0xCAFED00DUL : 0xD00DFACEUL);
                XorMix(ref h2, s.AnyButtonPressed ? 0x0BADC0DEUL : 0xDEADC0DEUL);
                XorMix(ref h1, s.LastAnyButtonPressed ? 0xABCDEF01UL : 0x10FEDCBAUL);
                XorMix(ref h2, s.LastAnyButtonPressed ? 0x13579BDFUL : 0xFDB97531UL);
            }

            // Entities: canonical order + sequential mix (breaks swap invariance)
            // Use the pre-sorted 'tmp' list so order is stable and reflects (type,pos,orientation).
            {
                const ulong P1 = 1099511628211UL; // FNV prime
                const ulong P2 = 1469598103934665603UL; // alternate prime
                foreach (var e in tmp)
                {
                    ulong v = 0;
                    v ^= (ulong)(byte)e.Type;
                    v ^= (ulong)(((uint)e.Pos.x << 16) ^ (uint)e.Pos.y);
                    v ^= (ulong)(byte)e.Orientation << 24;
                    // cascade mix so order matters (and is stable via canonical sort)
                    h1 ^= Mix64(v + 0xBADC0FFEE0DDF00DUL);
                    h1 *= P1;
                    h2 ^= Mix64((v * 1315423911UL) + 0xC001D00DDEADBEEFUL);
                    h2 *= P2;
                }
            }

            // Tile signature at togglable cells (captures swap state)
            foreach (var t in ctx.Togglables)
            {
                ref var cell = ref s.Grid.CellRef(t.Pos);
                var tt = cell.Type;
                byte ori = (byte)(cell.Orientation.HasValue ? ((byte)cell.Orientation.Value + 1) : 0);
                ulong pv = ((ulong)(uint)t.Pos.x << 32) ^ (ulong)(uint)t.Pos.y;
                XorMix(ref h1, pv ^ (((ulong)(byte)tt * 0xD6E8FEB9UL) ^ (ulong)ori));
                XorMix(ref h2, pv ^ (((ulong)(byte)tt * 0xA0761D64UL) ^ ((ulong)ori * 11400714819323198485UL)));
            }

            // Full-grid tile signature (strict) including orientation — ensures no pad-dependent merges
            for (int y = 0; y < s.Grid.H; y++)
                for (int x = 0; x < s.Grid.W; x++)
                {
                    ref var c = ref s.Grid.CellRef(new V2(x, y));
                    var tt = c.Type;
                    byte ori = (byte)(c.Orientation.HasValue ? ((byte)c.Orientation.Value + 1) : 0);
                    ulong pv = ((ulong)(uint)x << 32) ^ (ulong)(uint)y;
                    XorMix(ref h1, pv ^ (((ulong)(byte)tt * 0x9E3779B1UL) ^ (ulong)ori));
                    XorMix(ref h2, pv ^ (((ulong)(byte)tt * 0xC2B2AE3DUL) ^ ((ulong)ori * 1315423911UL)));
                }

            return new StateKey(h1, h2);
        }
    }
}
